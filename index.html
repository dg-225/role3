📊 구글 스프레드시트 (Apps Script) 연동

💡
구글 스프레드시트 연동 안내

구글 스프레드시트의 확장 프로그램 > Apps Script에서 웹 앱(Web App)으로 배포한 URL을 입력하면 실시간으로 데이터를 불러오거나 결과를 기록할 수 있습니다.

배포 시 액세스 권한을 '모든 사용자(Anyone)'로 설정해야 웹 브라우저에서 직접 연동됩니다.
📜 Google Apps Script 복사용 코드 보기

스프레드시트 Apps Script 편집기에 아래 코드를 붙여넣고 [배포] > [새 배포] > [웹 앱]으로 배포하세요.

// Google Apps Script 코드 예시
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    rows: data
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var body = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  sheet.clear();
  sheet.appendRow(["번호", "역할명", "담당학생", "역할미션", "배정일시"]);
  body.assignments.forEach(function(item, idx) {
    sheet.appendRow([idx + 1, item.role, item.student, item.desc, new Date()]);
  });
  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    count: body.assignments.length
  })).setMimeType(ContentService.MimeType.JSON);
}
